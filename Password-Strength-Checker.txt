//Password Strength Checker
//C++ Source Code
#include <iostream>
#include <string>

using namespace std;

int main()
{
    string password;

    bool hasUppercase = false;
    bool hasLowercase = false;
    bool hasNumber = false;
    bool hasSpecial = false;

    int score = 0;

    cout << "======================================" << endl;
    cout << "      PASSWORD STRENGTH CHECKER       " << endl;
    cout << "======================================" << endl;

    cout << "\nEnter your password: ";
    cin >> password;

    // Check every character of password
    for (int i = 0; i < password.length(); i++)
    {
        if (password[i] >= 'A' && password[i] <= 'Z')
        {
            hasUppercase = true;
        }
        else if (password[i] >= 'a' && password[i] <= 'z')
        {
            hasLowercase = true;
        }
        else if (password[i] >= '0' && password[i] <= '9')
        {
            hasNumber = true;
        }
        else
        {
            hasSpecial = true;
        }
    }

    // Calculate password score
    if (password.length() >= 8)
        score++;

    if (hasUppercase)
        score++;

    if (hasLowercase)
        score++;

    if (hasNumber)
        score++;

    if (hasSpecial)
        score++;

    // Password analysis
    cout << "\n======================================" << endl;
    cout << "         PASSWORD ANALYSIS            " << endl;
    cout << "======================================" << endl;

    cout << "\nPassword Length: " << password.length() << endl;

    cout << "Uppercase Letter: ";
    if (hasUppercase)
        cout << "Yes" << endl;
    else
        cout << "No" << endl;

    cout << "Lowercase Letter: ";
    if (hasLowercase)
        cout << "Yes" << endl;
    else
        cout << "No" << endl;

    cout << "Number: ";
    if (hasNumber)
        cout << "Yes" << endl;
    else
        cout << "No" << endl;

    cout << "Special Character: ";
    if (hasSpecial)
        cout << "Yes" << endl;
    else
        cout << "No" << endl;

    // Final result
    cout << "\n======================================" << endl;
    cout << "        PASSWORD STRENGTH RESULT      " << endl;
    cout << "======================================" << endl;

    if (score <= 2)
    {
        cout << "\nPassword Strength: WEAK" << endl;
        cout << "Your password is not secure." << endl;

        cout << "\nSuggestions:" << endl;

        if (password.length() < 8)
            cout << "- Use at least 8 characters." << endl;

        if (!hasUppercase)
            cout << "- Add at least one uppercase letter." << endl;

        if (!hasLowercase)
            cout << "- Add at least one lowercase letter." << endl;

        if (!hasNumber)
            cout << "- Add at least one number." << endl;

        if (!hasSpecial)
            cout << "- Add a special character like @, #, $, or !." << endl;
    }
    else if (score <= 4)
    {
        cout << "\nPassword Strength: MEDIUM" << endl;
        cout << "Your password is reasonably secure but can be improved." << endl;

        cout << "\nSuggestions:" << endl;

        if (password.length() < 8)
            cout << "- Use at least 8 characters." << endl;

        if (!hasUppercase)
            cout << "- Add at least one uppercase letter." << endl;

        if (!hasLowercase)
            cout << "- Add at least one lowercase letter." << endl;

        if (!hasNumber)
            cout << "- Add at least one number." << endl;

        if (!hasSpecial)
            cout << "- Add a special character like @, #, $, or !." << endl;
    }
    else
    {
        cout << "\nPassword Strength: STRONG" << endl;
        cout << "Excellent! Your password meets all security requirements." << endl;
    }

    cout << "\n======================================" << endl;
    cout << "     Thank you for using the program  " << endl;
    cout << "======================================" << endl;

    return 0;
}
